    while !todo.empty() {
      todo.pull()()
    }

    return document
  }

  def load<type V>($yaml::doc, $guid, &object): !yaml::doc {

  }

  def T::guidof($yaml::doc, $guid, type_t, type_t, *void): !?(guid|(guid, guid))
  def T::lookup($yaml::doc, $guid, type_t, type_t, (guid|(guid, guid))): !?*void

  private def inhume<type V>($pal::guid, todo: &[fun<void>], &!yaml::doc, object: &V): void {
    struct {
      val node = yaml::node::new_obj()

      pb::bind<V>(this)

      def pb_method<R, A...>() ...

      def pb_field<F>(const char*, get: (&V): !&F, set: (&V, F): !void)
          &val = get(object)
          if F::is_atomic || yaml
              node.put(name, val)
          else:
              guid = guidof(val)
              node.put(name, guid)
              todo += [val, guid]
              {
                  if guid in doc

                  inhume(guid, todo, doc, val)
